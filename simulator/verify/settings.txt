# Settings screen: Home -> Settings nav card -> Display block (brightness slider +
# 16/24-bit color toggle). Lower the brightness via the slider, then switch the
# color depth (which pops the restart-required modal). The script never confirms
# Restart — that would exit(0) the simulator.
settle
capture simulator/verify/out/settings_home.jpg
tap 589 1170
settle
capture simulator/verify/out/settings_open.jpg
# Drag the brightness slider down to ~25% (slider spans the card width at y~418).
tap 200 418
settle
capture simulator/verify/out/settings_brightness.jpg
# Extremes: grab the knob and drag past each end (the slider clamps to 0%/100%)
# so the knob must not clip. Waits let the drag samples register.
down 200 418
wait 60
move 40 418
wait 60
up
settle
capture simulator/verify/out/settings_min.jpg
down 200 418
wait 60
move 700 418
wait 60
up
settle
capture simulator/verify/out/settings_max.jpg
# Switch color depth to 16-bit -> half turns blue + the restart-required modal pops.
tap 511 484
settle
capture simulator/verify/out/settings_color.jpg
quit
